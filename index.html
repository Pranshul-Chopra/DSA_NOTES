Trees — Introduction & Terminology

THEORY

What is a Tree?

A tree is a non-linear, hierarchical data structure. Each element is called a node, and nodes are connected by edges. Every tree has exactly one root (topmost node), every non-root node has exactly one parent, and there is exactly one path from root to any node. A tree with n nodes has exactly n − 1 edges.

    int data;
    struct Node* left;
    struct Node* right;

struct Node* createNode(int val) {
    struct Node* newNode = (struct Node*)malloc(sizeof(struct Node));
    newNode->data  = val;
    newNode->left  = NULL;
    newNode->right = NULL;
    return newNode;

Level-Order Insertion (Linked List BT)

struct Node* insert(struct Node* root, int value) {
    struct Node* newNode = createNode(value);
    if (root == NULL) return newNode;

    struct Node* queue[100];
    int front = 0, rear = 0;
    queue[rear++] = root;

    while (front < rear) {
        struct Node* temp = queue[front++];

        if (temp->left == NULL)  { temp->left  = newNode; return root; }
        else                      queue[rear++] = temp->left;

        if (temp->right == NULL) { temp->right = newNode; return root; }
        else                      queue[rear++] = temp->right;
    }
    return root;
}

Tree Traversals (Recursive)

// Inorder: LEFT → ROOT → RIGHT  (gives sorted output for BST)
void inorder(struct Node* root) {
    if (root == NULL) return;
    inorder(root->left);
    printf("%d ", root->data);
    inorder(root->right);
}

// Preorder: ROOT → LEFT → RIGHT  (prefix expression, copy tree)
void preorder(struct Node* root) {
    if (root == NULL) return;
    printf("%d ", root->data);
    preorder(root->left);
    preorder(root->right);
}

// Postorder: LEFT → RIGHT → ROOT  (delete tree, postfix expression)
void postorder(struct Node* root) {
    if (root == NULL) return;
    postorder(root->left);
    postorder(root->right);
    printf("%d ", root->data);
}

Binary Search Trees (BST)

THEORY

The BST Property

For every node N: all keys in the left subtree < N < all keys in the right subtree. This must hold at every ancestor, not just immediate parent. An in-order traversal of a BST always gives a sorted sequence.

  • Insertion: New node is always a leaf. Walk down comparing until you fall off — hang node there.
  • Deletion Case 1 (leaf): Just detach.
  • Deletion Case 2 (one child): Parent points directly to the child (bypass deleted node).
  • Deletion Case 3 (two children): Replace with in-order successor (right once, then left as far as possible), then delete the successor (which is Case 1 or 2).
⚠ BST operations are O(h), NOT O(log n). They're only O(log n) when height is controlled (balanced). Inserting sorted data → skewed tree → O(n).

CODE

Search (Iterative + Recursive)

// Iterative Search — O(h) time, O(1) space
struct Node* searchIter(struct Node* root, int key) {
    while (root != NULL && root->data != key) {
        root = (key < root->data) ? root->left : root->right;
    }
    return root; // NULL if not found
}

// Recursive Search — O(h) time, O(h) space
struct Node* searchRec(struct Node* root, int key) {
    if (root == NULL || root->data == key) return root;
    if (key < root->data) return searchRec(root->left, key);
    return searchRec(root->right, key);
}

Insertion (Iterative + Recursive)

// Iterative Insert — O(h) time, O(1) space
struct Node* insertIter(struct Node* root, int key) {
    struct Node* newNode = createNode(key);
    if (root == NULL) return newNode;

    struct Node* parent = NULL;
    struct Node* current = root;

    while (current != NULL) {
        parent = current;
        current = (key < current->data) ? current->left : current->right;
    }

    if (key < parent->data) parent->left  = newNode;
    else                     parent->right = newNode;
    return root;
}

// Recursive Insert — O(h) time, O(h) space
struct Node* insertRec(struct Node* root, int key) {
    if (root == NULL) return createNode(key);
    if (key < root->data) root->left  = insertRec(root->left,  key);
    else                   root->right = insertRec(root->right, key);
    return root;
}

Deletion (Recursive)

// Helper: find minimum node in a subtree
struct Node* findMin(struct Node* node) {
    while (node->left != NULL) node = node->left;
    return node;
}

// Recursive Delete — O(h) time, O(h) space
struct Node* deleteRec(struct Node* root, int key) {
    if (root == NULL) return NULL;

    if      (key < root->data) root->left  = deleteRec(root->left,  key);
    else if (key > root->data) root->right = deleteRec(root->right, key);
    else {                          // Found the node
        // Case 1 & 2: 0 or 1 child
        if (root->left  == NULL) { struct Node* t = root->right; free(root); return t; }
        if (root->right == NULL) { struct Node* t = root->left;  free(root); return t; }
        // Case 3: 2 children — replace with in-order successor
        struct Node* successor = findMin(root->right);
        root->data  = successor->data;
        root->right = deleteRec(root->right, successor->data);
    }
    return root;
}

AVL Trees

THEORY

What is an AVL Tree?

An AVL Tree is a self-balancing BST (Adelson-Velsky & Landis, 1962). For every node, the heights of the left and right subtrees differ by at most 1. This guarantees O(log n) height always.

Insertion/Deletion proceed like a BST, but are followed by rotations to restore balance if needed.

FORMULAS
BALANCE FACTOR (BF)
BF(node) = height(left subtree) − height(right subtree)

BF = +1 → Left-Heavy (valid)
BF = 0 → Balanced (valid)
BF = −1 → Right-Heavy (valid)
|BF| > 1 → UNBALANCED → rotation needed

Height guarantee: h = O(log n)
MIN NODES FOR HEIGHT h
N(1) = 1, N(2) = 2
N(h) = 1 + N(h−1) + N(h−2) (like Fibonacci)

Conversely: max height for n nodes ≈ 1.44 log₂(n)
OperationTime ComplexitySpace
SearchO(log n)O(1) iter / O(log n) rec
InsertO(log n)O(log n)
DeleteO(log n)O(log n)
RotationO(1)O(1)
THEORY

The Four Rotations

The imbalanced node is called the critical node (A). Identify which subtree the new node is in relative to A.
Imbalance TypeConditionRotationSteps
LL New node in Left subtree of Left child of A (BF > +1) Single Right Rotation at A B becomes root; A becomes B's right child; B's right subtree becomes A's left child
RR New node in Right subtree of Right child of A (BF < −1) Single Left Rotation at A B becomes root; A becomes B's left child; B's left subtree becomes A's right child
LR New node in Right subtree of Left child of A Double: Left at B, then Right at A First rotate left on B (left child of A), then rotate right on A
RL New node in Left subtree of Right child of A Double: Right at B, then Left at A First rotate right on B (right child of A), then rotate left on A
Quick memory trick: LL → opposite rotation (Right). RR → opposite rotation (Left). LR/RL → double (two rotations).

Deletion Rotation Guide

BF of critical ABF of child BRotation
+2 (left heavy)0R0 — single right at A
+2 (left heavy)+1R1 — single right at A
+2 (left heavy)−1R(−1) — double: left at B, right at A
−2 (right heavy)0L0 — single left at A
−2 (right heavy)−1L(−1) — single left at A
−2 (right heavy)+1L1 — double: right at B, left at A
⚠ In deletion, unlike insertion, multiple ancestors may become unbalanced — you may need to apply rotations all the way up to the root.
CODE

AVL Node + Helper Functions

struct Node {
    int key, height;
    struct Node *left, *right;
};

int height(struct Node* n) { return n ? n->height : 0; }
int max(int a, int b) { return a > b ? a : b; }
int getBalance(struct Node* n) { return n ? height(n->left) - height(n->right) : 0; }

// Right Rotation (for LL imbalance)
struct Node* rotateRight(struct Node* y) {
    struct Node* x  = y->left;
    struct Node* T2 = x->right;
    x->right = y; y->left = T2;
    y->height = max(height(y->left), height(y->right)) + 1;
    x->height = max(height(x->left), height(x->right)) + 1;
    return x;
}

// Left Rotation (for RR imbalance)
struct Node* rotateLeft(struct Node* x) {
    struct Node* y  = x->right;
    struct Node* T2 = y->left;
    y->left = x; x->right = T2;
    x->height = max(height(x->left), height(x->right)) + 1;
    y->height = max(height(y->left), height(y->right)) + 1;
    return y;
}

// AVL Insert
struct Node* insert(struct Node* node, int key) {
    if (!node) return newNode(key);
    if      (key < node->key) node->left  = insert(node->left,  key);
    else if (key > node->key) node->right = insert(node->right, key);
    else return node; // No duplicates

    node->height = 1 + max(height(node->left), height(node->right));
    int bf = getBalance(node);

    if (bf > 1  && key < node->left->key)  return rotateRight(node);      // LL
    if (bf < -1 && key > node->right->key) return rotateLeft(node);       // RR
    if (bf > 1  && key > node->left->key)  { node->left = rotateLeft(node->left);  return rotateRight(node); } // LR
    if (bf < -1 && key < node->right->key) { node->right = rotateRight(node->right); return rotateLeft(node);  } // RL
    return node;
}

Red-Black Trees

THEORY

What is a Red-Black Tree?

A Red-Black Tree is a self-balancing BST where each node stores an extra attribute: color (Red or Black). It is less strictly balanced than AVL but faster for frequent insertions/deletions because it requires fewer rotations (at most 2 per insert vs up to O(log n) for AVL).

The 5 Properties (Must Know)

  1. Every node is either Red or Black.
  2. The root is always Black.
  3. Every leaf (NULL) is Black.
  4. If a node is Red, both its children are Black (no two consecutive red nodes).
  5. For any node, all paths from that node to descendant leaves have the same number of Black nodes (Black-Height).
FORMULAS
HEIGHT GUARANTEE
Height h ≤ 2 log₂(n + 1) = O(log n)

Black-Height (bh) = number of black nodes on any path from root to leaf (not counting root)

Minimum nodes for black-height bh = 2^bh − 1
OperationTime Complexityvs AVL
SearchO(log n)Similar (AVL slightly faster due to stricter balance)
InsertO(log n)RB faster (≤ 2 rotations vs more for AVL)
DeleteO(log n)RB faster (≤ 3 rotations)

AVL vs Red-Black — When to Use Which

CriterionAVL TreeRed-Black Tree
Balance strictnessStrict (BF ≤ 1)Relaxed (height ≤ 2 log n)
Search speedSlightly fasterSlightly slower
Insert/Delete speedSlowerFaster
Best forRead-heavy workloadsFrequent writes (most practical uses)
Used inDatabases (read-heavy)Linux kernel, C++ std::map/set, Java TreeMap
THEORY

Insertion Cases (after inserting as Red)

New node is always inserted as Red. If the parent is Black, done. If parent is Red (violation of Property 4), fix using:

  • Case 1 — Uncle is Red: Recolor parent + uncle to Black, grandparent to Red. Move up to grandparent and repeat.
  • Case 2 — Uncle is Black, node is inner grandchild: Rotate parent in opposite direction → converts to Case 3.
  • Case 3 — Uncle is Black, node is outer grandchild: Rotate grandparent, swap colors of parent and grandparent.

At most 2 rotations needed per insertion. Always ensure root stays Black at the end.

B-Trees

THEORY

Why B-Trees?

B-Trees are designed for disk/external storage where reading a single node costs an entire block access. By storing many keys per node, B-Trees minimize disk accesses (height is very small even for millions of records).

A B-Tree of order m (minimum degree t) satisfies: each node holds between t−1 and 2t−1 keys, and between t and 2t children (except root which can have as few as 1 key).

B-Tree Properties (Order m / Min-Degree t)

  • All leaves are at the same level (perfectly balanced)
  • Each non-root node has at least ⌈m/2⌉ − 1 keys (minimum degree t: at least t−1 keys)
  • Each node has at most m − 1 keys (or 2t−1 keys)
  • A node with k keys has k + 1 children
  • Keys within a node are sorted
  • All leaf nodes are at the same depth
FORMULAS
B-TREE HEIGHT & COMPLEXITY
Height h ≤ log_t ⌈(n+1)/2⌉ ≈ O(log_t n)

Number of disk accesses per operation ≈ O(h) = O(log_t n)

Each node of order m:
Max keys per node = m − 1
Min keys (non-root) = ⌈m/2⌉ − 1
Max children = m
Min children (non-root internal) = ⌈m/2⌉
OperationTime ComplexityDisk Accesses
SearchO(t · log_t n)O(log_t n)
InsertO(t · log_t n)O(log_t n)
DeleteO(t · log_t n)O(log_t n)
THEORY

Insertion Algorithm

  1. Find the appropriate leaf node using search.
  2. Insert the key in sorted order within that node.
  3. If node does not overflow → done.
  4. If node overflows (has m keys): split at the median key.
  5. Push median key up to parent; left and right halves become two nodes.
  6. If parent also overflows → repeat upward. If root splits → new root (height increases by 1).
Split example (order 5, node [10|20|30|40|50]): median = 30. Promote 30 to parent. Left child = [10|20], Right child = [40|50].

Deletion Algorithm

  • Key in leaf: Simply delete.
  • Key in internal node: Replace with in-order predecessor or successor (from a leaf), then delete from leaf.
  • Underflow (node has t−1 keys after deletion): Borrow from sibling (through parent), or merge with sibling (pull parent key down).
  • Merging may propagate upward, possibly reducing tree height.

Heaps & Heap Sort

THEORY

What is a Heap?

A Heap is a complete binary tree satisfying the heap property:

  • Max-Heap: Parent ≥ both children → largest element always at root.
  • Min-Heap: Parent ≤ both children → smallest element always at root.

Heaps are always stored as arrays (efficient since they're complete binary trees). The largest element of a max-heap is always at index 0 (or index 1 for 1-based).

FORMULAS
HEAP ARRAY INDEXING (0-based)
Left child = 2i + 1
Right child = 2i + 2
Parent = ⌊(i − 1) / 2⌋

Leaves are at indices ⌊n/2⌋ to n−1
BUILD-MAX-HEAP starts heapify from ⌊n/2⌋ − 1 downto 0
OperationTimeNote
MAX-HEAPIFYO(log n)Sift down one element
BUILD-MAX-HEAPO(n)Not O(n log n)! Tighter analysis gives O(n)
HEAP-SORTO(n log n)O(n) build + n × O(log n) extractions
Insert into heapO(log n)Insert at end, bubble up
Extract-MaxO(log n)Remove root, replace with last, heapify down
Get-MaxO(1)Root is always max
SpaceO(1) extraIn-place sort (no extra array needed)
⚠ BUILD-MAX-HEAP is O(n) NOT O(n log n). Heap sort overall is O(n log n) and sorts in-place (unlike merge sort which needs extra space).
CODE

MAX-HEAPIFY + BUILD-MAX-HEAP + HEAPSORT

// MAX-HEAPIFY — restore heap property at index i, heap size n
// Assumes left[i] and right[i] are valid max-heaps
void maxHeapify(int A[], int i, int n) {
    int left    = 2*i + 1;
    int right   = 2*i + 2;
    int largest = i;

    if (left  < n && A[left]  > A[largest]) largest = left;
    if (right < n && A[right] > A[largest]) largest = right;

    if (largest != i) {
        // swap A[i] and A[largest]
        int tmp = A[i]; A[i] = A[largest]; A[largest] = tmp;
        maxHeapify(A, largest, n);  // recurse on displaced element
    }
}

// BUILD-MAX-HEAP — O(n)
// Start from last internal node (index n/2 - 1) going to root
void buildMaxHeap(int A[], int n) {
    for (int i = n/2 - 1; i >= 0; i--)
        maxHeapify(A, i, n);
}

// HEAP SORT — O(n log n), in-place
void heapSort(int A[], int n) {
    buildMaxHeap(A, n);               // Step 1: build heap
    for (int i = n - 1; i >= 1; i--) { // Step 2: repeatedly extract max
        // swap root (max) with last element
        int tmp = A[0]; A[0] = A[i]; A[i] = tmp;
        maxHeapify(A, 0, i);           // restore heap on reduced array
    }
    // Array is now sorted ascending
}

Heap Sort Step-by-Step (example: [4,10,3,5,1])

Step 1 — Build max-heap: Start from index 1 (n/2 - 1 = 1).

  • Heapify(1): 10 > 4, swap → [4,10,3,5,1] → compare 4 with 5 → swap → [10,5,3,4,1]
  • Heapify(0): 10 already largest → [10,5,3,4,1] ✓ (max-heap built)

Step 2 — Extract max repeatedly:

  • Swap A[0]↔A[4]: [1,5,3,4, 10] → heapify(0,4) → [5,4,3,1, 10]
  • Swap A[0]↔A[3]: [1,4,3, 5,10] → heapify(0,3) → [4,1,3, 5,10]
  • Swap A[0]↔A[2]: [3,1, 4,5,10] → heapify(0,2) → [3,1, 4,5,10]
  • Swap A[0]↔A[1]: [1,3,4,5,10] → Done! Sorted: [1,3,4,5,10]

Graphs, BFS & DFS

THEORY

Graph Basics

A Graph G = (V, E) is a set of vertices V and edges E. Unlike trees, graphs can have cycles and disconnected components.

  • Undirected graph: edges have no direction
  • Directed graph (Digraph): edges have direction; (u,v) ≠ (v,u)
  • Weighted graph: each edge has a value/weight
  • Dense graph: |E| ≈ |V|²
  • Sparse graph: |E| << |V|²
FORMULAS
GRAPH EDGE COUNTS
Max edges in undirected graph with V vertices = V(V−1)/2
Max edges in directed graph with V vertices = V(V−1)

Adjacency Matrix space: O(V²)
Adjacency List space: O(V + E)

CODE

BFS Algorithm

// BFS from source s — finds shortest paths (unweighted)
// Time: O(V+E),  Space: O(V) for queue + color arrays
void BFS(Graph G, int s) {
    // Initialize all vertices
    for (int u = 0; u < G.V; u++) {
        color[u] = WHITE;
        d[u]     = INF;      // distance from s
        pi[u]    = -1;      // predecessor
    }
    color[s] = GRAY;
    d[s]     = 0;

    Queue Q;
    enqueue(Q, s);

    while (!isEmpty(Q)) {
        int u = dequeue(Q);
        for each v in Adj[u] {
            if (color[v] == WHITE) {
                color[v] = GRAY;
                d[v]     = d[u] + 1;  // one hop further
                pi[v]    = u;
                enqueue(Q, v);
            }
        }
        color[u] = BLACK;  // fully explored
    }
}
// After BFS: d[v] = shortest path length from s to v
// pi[v]    = parent in BFS tree (reconstruct path by tracing back)

DFS Algorithm

// DFS — Time: O(V+E),  Space: O(V) for recursion stack
int time = 0;

void DFS(Graph G) {
    for each vertex u in G.V {
        color[u] = WHITE;
        pi[u]    = -1;
    }
    time = 0;
    for each vertex u in G.V {
        if (color[u] == WHITE)
            DFS_VISIT(u);  // handles disconnected graphs
    }
}

void DFS_VISIT(int u) {
    color[u] = GRAY;
    d[u]     = ++time;   // discovery timestamp

    for each v in Adj[u] {
        if (color[v] == WHITE) {
            pi[v] = u;
            DFS_VISIT(v);
        }
    }
    color[u] = BLACK;
    f[u]     = ++time;   // finish timestamp
}
// d[u]/f[u] = discovery/finish times (useful for cycle detection, topological sort)

int adjMatrix[V][V] = {0};

// Adjacency List (for sparse graphs)
struct AdjNode { int dest; struct AdjNode* next; };

void addEdge(int u, int v) {
    struct AdjNode* node = malloc(sizeof(struct AdjNode));
    node->dest = v; node->next = adjList[u]; adjList[u] = node;

    node->dest = u; node->next = adjList[v]; adjList[v] = node;
